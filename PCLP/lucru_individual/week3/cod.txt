#  cod care scrie media aritmetica a 2 numere intregi, cu 3 zecimale ( am facut o sa verific ca stiu sa folosesc printf)
#include <stdio.h>

int main()
{

    int a, b;

    scanf("%d %d", &a, &b);

    float medie = (float)(a + b) / 2;

    printf("Media a doua numere intregi %d si %d cu 3 zecimale este: %.3f \n", a, b, medie);

    return 0;
}

#include <stdio.h>

int main() {
  int A, B;

  scanf("%d %d", &A, &B);

  for (int i = A; i <= B; i++) {
    bool prim = true;
    if (i == 1) {
      prim = false;
    } else if (i % 2 == 0 && i != 2) {
      prim = false;
    } else {
      for (int j = 3; j * j <= i; j += 2) {
        if (i % j == 0) {
          prim = false;
        }
      }
    }

    if (prim == true) {
      printf("%d ", i);
    }
  }

  printf("\n");
  return 0;
}

    al_doilea_for:

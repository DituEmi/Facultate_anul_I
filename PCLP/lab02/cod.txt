problema 4:

#include <stdio.h>

int main()
{
    int n;

    scanf("%d", &n);

    for (int i = 0; i < n; i++)
    {
        for (int j = n - i; j > 0; j--)
            printf("*");
        printf("\n");
    }

    return 0;
}

#include <stdio.h>

int main()
{
    int n;
    int aux, palin = 0;

    scanf("%d", &n);
    aux = n;

    while (n != 0)
    {
        palin = palin * 10 + n % 10;
        n = n / 10;
    }

    if (palin == aux)
        printf("DA\n");
    else
        printf("NU\n");

    return 0;
}
